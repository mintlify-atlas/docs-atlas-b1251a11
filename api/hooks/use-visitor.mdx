# useVisitor

React hook for accessing and managing the current visitor's identity and metadata.

## Import

```typescript
import { useVisitor } from '@cossistant/react';
```

## Basic Usage

```tsx
import { useVisitor } from '@cossistant/react';

function VisitorInfo() {
  const { visitor, identify, setVisitorMetadata } = useVisitor();
  
  if (!visitor) {
    return <div>No visitor session</div>;
  }
  
  return (
    <div>
      <p>Visitor ID: {visitor.id}</p>
      <p>Browser: {visitor.browser}</p>
      <p>Location: {visitor.city}, {visitor.country}</p>
    </div>
  );
}
```

## Examples

### Display Visitor Info

```tsx
function VisitorCard() {
  const { visitor } = useVisitor();
  
  if (!visitor) return null;
  
  return (
    <div className="visitor-card">
      <div className="device">
        {visitor.deviceType === 'mobile' ? '📱' : '💻'}
        {visitor.browser} on {visitor.os}
      </div>
      <div className="location">
        📍 {visitor.city}, {visitor.country}
      </div>
      {visitor.language && (
        <div className="language">
          🌐 {visitor.language}
        </div>
      )}
    </div>
  );
}
```

### Identify Visitor

```tsx
function LoginForm() {
  const { identify } = useVisitor();
  const [email, setEmail] = useState('');
  const [name, setName] = useState('');
  
  const handleSubmit = async (e) => {
    e.preventDefault();
    
    const result = await identify({
      email,
      name,
      externalId: 'user_123', // Your system's user ID
      metadata: {
        plan: 'pro',
        signupDate: new Date().toISOString()
      }
    });
    
    if (result) {
      console.log('Identified as contact:', result.contactId);
    }
  };
  
  return (
    <form onSubmit={handleSubmit}>
      <input
        type="email"
        value={email}
        onChange={(e) => setEmail(e.target.value)}
        placeholder="Email"
      />
      <input
        type="text"
        value={name}
        onChange={(e) => setName(e.target.value)}
        placeholder="Name"
      />
      <button type="submit">Identify</button>
    </form>
  );
}
```

### Update Metadata

```tsx
function UpdatePreferences() {
  const { setVisitorMetadata } = useVisitor();
  const [preferences, setPreferences] = useState({
    theme: 'light',
    notifications: true
  });
  
  const save = async () => {
    await setVisitorMetadata({
      preferences
    });
  };
  
  return (
    <div>
      <label>
        <input
          type="checkbox"
          checked={preferences.notifications}
          onChange={(e) => setPreferences(prev => ({
            ...prev,
            notifications: e.target.checked
          }))}
        />
        Enable notifications
      </label>
      <button onClick={save}>Save</button>
    </div>
  );
}
```

### Track User Actions

```tsx
function TrackPurchase({ orderId, amount }) {
  const { setVisitorMetadata } = useVisitor();
  
  useEffect(() => {
    setVisitorMetadata({
      lastPurchase: {
        orderId,
        amount,
        date: new Date().toISOString()
      },
      totalSpent: amount // Will be summed server-side
    });
  }, [orderId, amount, setVisitorMetadata]);
  
  return <div>Order confirmed</div>;
}
```

### Conditional Rendering Based on Visitor

```tsx
function LocalizedGreeting() {
  const { visitor } = useVisitor();
  
  const greeting = useMemo(() => {
    const lang = visitor?.language?.split('-')[0];
    
    const greetings = {
      'en': 'Hello',
      'es': 'Hola',
      'fr': 'Bonjour',
      'de': 'Guten Tag',
      'ja': 'こんにちは'
    };
    
    return greetings[lang] || 'Hello';
  }, [visitor?.language]);
  
  return <h1>{greeting}!</h1>;
}
```

### Show Returning Visitor Message

```tsx
function WelcomeBack() {
  const { visitor } = useVisitor();
  
  const isReturning = useMemo(() => {
    if (!visitor?.createdAt) return false;
    
    const created = new Date(visitor.createdAt);
    const hoursSinceCreation = (Date.now() - created.getTime()) / (1000 * 60 * 60);
    
    return hoursSinceCreation > 24; // Created more than 24 hours ago
  }, [visitor?.createdAt]);
  
  if (isReturning) {
    return <div className="welcome-back">Welcome back! 👋</div>;
  }
  
  return <div className="welcome-new">Welcome! 🎉</div>;
}
```

## Visitor vs Contact

### Visitor
- **Anonymous**: Created automatically on first visit
- **Temporary**: Tracked by browser localStorage
- **Device-specific**: One visitor per browser/device
- **Limited data**: Browser, device, location

### Contact
- **Identified**: Created via `identify()` with email/name
- **Persistent**: Survives across devices when logged in
- **Unified**: Multiple visitors can link to one contact
- **Rich data**: Custom metadata, conversation history

```tsx
// Visitor (anonymous)
{
  id: '01JG000000000000000000000',
  browser: 'Chrome',
  city: 'New York'
}

// After identify()
{
  // Visitor is linked to:
  contactId: '01JG000000000000000000005',
  email: 'user@example.com',
  name: 'John Doe',
  metadata: { plan: 'pro', ... }
}
```

## Error Handling

```tsx
function SafeIdentify() {
  const { identify } = useVisitor();
  const [error, setError] = useState(null);
  
  const handleIdentify = async (email) => {
    try {
      const result = await identify({ email });
      
      if (!result) {
        setError('Failed to identify visitor');
        return;
      }
      
      console.log('Success:', result);
    } catch (err) {
      setError(err.message);
    }
  };
  
  return (
    <div>
      {error && <div className="error">{error}</div>}
    </div>
  );
}
```

## TypeScript

```typescript
import { useVisitor } from '@cossistant/react';
import type { PublicVisitor, VisitorMetadata } from '@cossistant/types';

function TypedComponent() {
  const { visitor, setVisitorMetadata } = useVisitor();
  
  // Fully typed
  const visitorId: string | undefined = visitor?.id;
  const city: string | null | undefined = visitor?.city;
  
  // Metadata is type-safe
  const metadata: VisitorMetadata = {
    customField: 'value',
    nested: {
      property: 123
    }
  };
  
  setVisitorMetadata(metadata);
}
```
